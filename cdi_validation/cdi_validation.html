Compute CDI Onset Ages

For each word, the earliest age (months) at which the proportion of children reported to produce it crosses 0.1 / 0.15 / 0.2.

CDI_onset_age <- cdi_data %>%
  group_by(word) %>%
  summarise(
    age_above_0.1  = min(age[proportion >= 0.1],  na.rm = TRUE),
    age_above_0.15 = min(age[proportion >= 0.15], na.rm = TRUE),
    age_above_0.2  = min(age[proportion >= 0.2],  na.rm = TRUE)
  )

Normalize Word Tokens and Join

CDI and CHILDES tokenize a few words differently. We apply only unambiguous one-to-one mappings; contractions ('m/am, n't/don't) are structurally different tokens on the CHILDES side and are left unmatched rather than force-mapped.

CDI_onset_age_norm <- CDI_onset_age %>%
  mutate(word = recode(word,
    "can (auxiliary)" = "can",
    "inside/in"        = "in"
  ))

comparison <- inner_join(age_of_production, CDI_onset_age_norm, by = "word")

childes_unmatched <- setdiff(age_of_production$word, CDI_onset_age_norm$word)

Validation Statistics

thresholds <- c("age_above_0.1", "age_above_0.15", "age_above_0.2")

cor_results <- map_dfr(thresholds, function(t) {
  pear <- cor.test(comparison$Estimate, comparison[[t]], method = "pearson")
  spear <- cor.test(comparison$Estimate, comparison[[t]], method = "spearman")
  diff <- comparison$Estimate - comparison[[t]]
  tibble(
    cdi_threshold = t,
    pearson_r   = unname(pear$estimate),
    pearson_p   = pear$p.value,
    spearman_rho = unname(spear$estimate),
    spearman_p   = spear$p.value,
    mean_abs_diff = mean(abs(diff)),
    rmse = sqrt(mean(diff^2))
  )
})
